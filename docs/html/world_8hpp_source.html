304 template <typename C, typename... Components, typename N = none>
305 static std::vector<entity_t> query_components()
306 {
307 if (!world::components)
308 {
309 return {};
310 }
311
312 std::vector<entity_t> matched;
313
314 if (components->count(type_id<C>) == 0)
315 {
316 return matched;
317 }
318
319 for (auto component : components->at(type_id<C>))
320 {
321 matched.push_back(component->entity);
322 }
323
324 if (matched.empty())
325 return matched;
326 if (sizeof...(Components) == 0)
327 return matched;
328
329 query_components_rec<Components..., none>(&matched);
330
331 return matched;
332 }
333
334 template <typename C, typename... Components>
335 static void query_components_rec(std::vector<entity_t> *entities)
336 {
337 if (entities->empty())
338 return;
339 std::vector<entity_t> matched;
340
341 if (components->count(type_id<C>) == 0)
342 {
343 return;
344 }
345 for (auto component : components->at(type_id<C>))
346 {
347 if (std::find(entities->begin(), entities->end(),
348 (int &) component->entity)
349 != entities->end())
350 {
351 matched.push_back(component->entity);
352 }
353 }
354 *entities = matched;
355
356 if (sizeof...(Components) == 0)
357 return;
358
359 (query_components_rec<Components>(entities), ...);
360 }
