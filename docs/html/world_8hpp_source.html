326 template <typename C, typename... Components, typename N = none>
327 static std::vector<types::entity_id> query_components()
328 {
329 if (!world::components)
330 {
331 return {};
332 }
333
334 std::vector<types::entity_id> matched;
335
336 if (components->count(type_id<C>) == 0)
337 {
338 return matched;
339 }
340
341 for (auto component : components->at(type_id<C>))
342 {
343 matched.push_back(component->entity);
344 }
345
346 if (matched.empty())
347 return matched;
348 if (sizeof...(Components) == 0)
349 return matched;
350
351 query_components_rec<Components..., none>(&matched);
352
353 return matched;
354 }
355
356 template <typename C, typename... Components>
357 static void query_components_rec(std::vector<types::entity_id> *entities)
358 {
359 if (entities->empty())
360 return;
361 std::vector<types::entity_id> matched;
362
363 if (components->count(type_id<C>) == 0)
364 {
365 return;
366 }
367 for (auto component : components->at(type_id<C>))
368 {
369 if (std::find(entities->begin(), entities->end(),
370 (int &) component->entity)
371 != entities->end())
372 {
373 matched.push_back(component->entity);
374 }
375 }
376 *entities = matched;
377
378 if (sizeof...(Components) == 0)
379 return;
380
381 (query_components_rec<Components>(entities), ...);
382 }
